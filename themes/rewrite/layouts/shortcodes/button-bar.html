{{/* {{< button-bar "icon:text:link" "icon:text:link" … >}} Self-closing. Each positional "icon:text:link" triple renders as a primary button. Matches upstream API. Must NOT reference .Inner — Hugo's parser would then require a closing tag, which the content never provides. */}}
{{ range $.Params }} {{ $parts := split . ":" }} {{ $icon := index $parts 0 }} {{ $text := index $parts 1 }} {{ $link := index $parts 2 }} {{ $href := cond (hasPrefix $link "http") $link (absURL $link) }} {{ with $icon }}{{ end }} {{ $text }} {{ end }}
